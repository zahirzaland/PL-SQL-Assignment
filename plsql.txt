1. Rectangle Area Program

    width NUMBER := 5;
    area NUMBER;

    DBMS_OUTPUT.PUT_LINE('Area of Rectangle = ' || area);
END;

2. Find Average of Three Numbers

DECLARE
    a NUMBER := 10;
    b NUMBER := 20;
    c NUMBER := 30;
    avg_num NUMBER;
BEGIN
    avg_num := (a + b + c) / 3;

    DBMS_OUTPUT.PUT_LINE('Average = ' || avg_num);
END;

4. Find Largest Number among Three Numbers
DECLARE
    a NUMBER := 45;
    b NUMBER := 78;
    c NUMBER := 25;
BEGIN
    IF a > b AND a > c THEN
        DBMS_OUTPUT.PUT_LINE('Largest Number = ' || a);

    ELSIF b > a AND b > c THEN
        DBMS_OUTPUT.PUT_LINE('Largest Number = ' || b);

    ELSE
        DBMS_OUTPUT.PUT_LINE('Largest Number = ' || c);
    END IF;
END;

    END LOOP;

    DBMS_OUTPUT.PUT_LINE('Factorial = ' || fact);
END;
12. Electricity Bill Calculation Program

DECLARE
    units NUMBER := 120;
    bill NUMBER;
BEGIN
    IF units <= 100 THEN
        bill := units * 5;

    ELSE
        bill := units * 8;
    END IF;

    DBMS_OUTPUT.PUT_LINE('Electricity Bill = ' || bill);
END;
